import os
from dataclasses import dataclass


@dataclass(frozen=True)
class Settings:
    database_url: str = os.getenv("DATABASE_URL", "sqlite:///./jobdb.db")
    redis_host: str = os.getenv("REDIS_HOST", "localhost")
    redis_port: int = int(os.getenv("REDIS_PORT", "6379"))
    redis_db: int = int(os.getenv("REDIS_DB", "0"))
    redis_queue_name: str = os.getenv("REDIS_QUEUE_NAME", "job_queue")
    max_retries: int = int(os.getenv("JOB_MAX_RETRIES", "3"))
    processing_timeout_seconds: int = int(os.getenv("JOB_PROCESSING_TIMEOUT_SECONDS", "10"))
    worker_poll_interval_seconds: float = float(os.getenv("JOB_WORKER_POLL_INTERVAL_SECONDS", "2"))
    recovery_sleep_seconds: int = int(os.getenv("JOB_RECOVERY_SLEEP_SECONDS", "360"))


settings = Settings()
